public class Main {
    public static void main(String[] args) {

        Scanner ler = new Scanner(System.in);

        int idade, contador = 0;
        double altura, soma_alturas = 0;

        for (int i = 1; i <= 20; i++) {

            System.out.print("Digite a idade: ");
            idade = ler.nextInt();

            System.out.print("Digite a altura: ");
            altura = ler.nextDouble();

            if (idade > 50) {
                soma_alturas += altura;
                contador++;
            }
        }

        if (contador > 0) {
            double media_alturas = soma_alturas / contador;
            System.out.println("Media das alturas: " + media_alturas);
        } else {
            System.out.println("Nao ha pessoas com mais de 50 anos.");
        }
    }
}

public class Main {
    public static void main(String[] args) {

        Scanner ler = new Scanner(System.in);

        int idade, opiniao;
        int soma_idades_otimo = 0, qtd_otimo = 0;
        int qtd_regular = 0, qtd_bom = 0;

        for (int i = 1; i <= 15; i++) {

            System.out.print("Digite a idade: ");
            idade = ler.nextInt();

            System.out.print("Digite a opiniao (3-otimo, 2-bom, 1-regular): ");
            opiniao = ler.nextInt();

            if (opiniao == 3) {
                soma_idades_otimo += idade;
                qtd_otimo++;
            } else if (opiniao == 2) {
                qtd_bom++;
            } else if (opiniao == 1) {
                qtd_regular++;
            }
        }

        if (qtd_otimo > 0) {
            double media_idades_otimo = (double) soma_idades_otimo / qtd_otimo;
            System.out.println("Media das idades (otimo): " + media_idades_otimo);
        } else {
            System.out.println("Nao houve respostas otimo");
        }

        System.out.println("Quantidade de regular: " + qtd_regular);

        double porcentagem_bom = (qtd_bom * 100.0) / 15;
        System.out.println("Porcentagem de bom: " + porcentagem_bom + "%");
    }
}

public class Main {
    public static void main(String[] args) {

        Scanner ler = new Scanner(System.in);

        int idade, qtd_c = 0, qtd_s = 0, qtd_v = 0, qtd_d = 0, soma_idade_v = 0;
        char estado;

        for (int i = 1; i <= 20; i++) {

            System.out.print("Idade: ");
            idade = ler.nextInt();

            System.out.print("Estado civil (C/S/V/D): ");
            estado = ler.next().charAt(0);

            if (estado == 'C' || estado == 'c') {
                qtd_c++;
            } else if (estado == 'S' || estado == 's') {
                qtd_s++;
            } else if (estado == 'V' || estado == 'v') {
                qtd_v++;
                soma_idade_v += idade;
            } else if (estado == 'D' || estado == 'd') {
                qtd_d++;
            }
        }

        System.out.println("Casados: " + qtd_c);
        System.out.println("Solteiros: " + qtd_s);

        if (qtd_v > 0) {
            double media = (double) soma_idade_v / qtd_v;
            System.out.println("Media idades viuvos: " + media);
        }

        double porcentagem = (qtd_d * 100.0) / 20;
        System.out.println("Porcentagem desquitados: " + porcentagem + "%");
    }
}

public class Main {
    public static void main(String[] args) {

        Scanner ler = new Scanner(System.in);

        int num, maior = 0, menor = 0;
        boolean primeiro = true;

        while (true) {

            System.out.print("Numero: ");
            num = ler.nextInt();

            if (num == 0) break;

            if (num < 0) {
                System.out.println("Valor negativo!");
                continue;
            }

            if (primeiro) {
                maior = num;
                menor = num;
                primeiro = false;
            } else {
                if (num > maior) maior = num;
                if (num < menor) menor = num;
            }
        }

        if (!primeiro) {
            System.out.println("Maior: " + maior);
            System.out.println("Menor: " + menor);
        }
    }
}
